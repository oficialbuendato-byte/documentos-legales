function mostrarUsuario(usuario) { const vistaInicio = document.getElementById('vista-inicio'); const vistaUsuario = document.getElementById('vista-usuario'); const nombreUsuario = document.getElementById('nombre-usuario'); const avatarUsuario = document.getElementById('avatar-usuario'); if (nombreUsuario) { nombreUsuario.textContent = usuario.display_name || 'Usuario de TikTok'; } if (avatarUsuario && usuario.avatar_url) { avatarUsuario.src = usuario.avatar_url; avatarUsuario.style.display = 'block'; } if (vistaInicio) { vistaInicio.style.display = 'none'; } if (vistaUsuario) { vistaUsuario.style.display = 'block'; } } function cargarUsuarioGuardado() { const usuarioGuardado = localStorage.getItem('tiktok_usuario'); if (!usuarioGuardado) { return; } try { const usuario = JSON.parse(usuarioGuardado); mostrarUsuario(usuario); } catch (error) { console.error('Error leyendo el usuario guardado:', error); localStorage.removeItem('tiktok_usuario'); } } document.addEventListener('DOMContentLoaded', function () { cargarUsuarioGuardado(); });
